#include <Arduino_FreeRTOS.h>

// Two independent periodic tasks demonstrate fixed-priority preemption.
// Task2 (priority 2) has higher priority than Task1 (priority 1).
void Task1(void *pvParameters);
void Task2(void *pvParameters);

void setup()
{
    Serial.begin(9600);
    pinMode(7, OUTPUT);
    pinMode(8, OUTPUT);
    digitalWrite(7, LOW);
    digitalWrite(8, LOW);
    Serial.println("This is setup function");

    BaseType_t task1Result = xTaskCreate(Task1, "Task1", 128, NULL, 1, NULL);
    BaseType_t task2Result = xTaskCreate(Task2, "Task2", 128, NULL, 2, NULL);
    if (task1Result != pdPASS || task2Result != pdPASS) {
        Serial.println("ERROR: task creation failed");
        for (;;) { }
    }

    vTaskStartScheduler();
    Serial.println("ERROR: scheduler failed to start");
    for (;;) { }
}

void loop()
{
    // The scheduler owns execution after successful startup.
}

void Task1(void *pvParameters)
{
    (void)pvParameters;
    for (;;) {
        digitalWrite(7, HIGH);
        Serial.println("Task-1 is running");
        vTaskDelay(pdMS_TO_TICKS(1000));
        digitalWrite(7, LOW);
    }
}

void Task2(void *pvParameters)
{
    (void)pvParameters;
    for (;;) {
        digitalWrite(8, HIGH);
        Serial.println("Task-2 is running");
        vTaskDelay(pdMS_TO_TICKS(1000));
        digitalWrite(8, LOW);
    }
}
